cmake_minimum_required(VERSION 3.25)
project(ckvision_log_clients LANGUAGES CXX)
find_package(ckvision CONFIG REQUIRED)
foreach(client IN ITEMS file_monitor recipes file_contract)
    add_executable(log_${client} ${client}.cpp)
    target_link_libraries(log_${client} PRIVATE ckvision::cvision)
    target_compile_features(log_${client} PRIVATE cxx_std_20)
    if(MSVC)
        target_compile_options(log_${client} PRIVATE /W4 /WX /permissive- /utf-8)
    else()
        target_compile_options(log_${client} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
if(MINGW)
    target_link_options(log_file_monitor PRIVATE -municode)
endif()
enable_testing()
add_test(NAME log_recipes COMMAND log_recipes)
# Caller supplies an explicitly selected, disposable absolute scratch directory.
if(DEFINED LOG_CLIENT_SCRATCH)
    add_test(NAME log_file_contract COMMAND log_file_contract "${LOG_CLIENT_SCRATCH}")
    set_tests_properties(log_file_contract PROPERTIES TIMEOUT 60)
endif()
